# ── Helper: parse uploaded file ────────────────────────────────────────────────

def parse_upload(contents, filename):
    """Decode base64 upload → DataFrame or None."""
    if contents is None:
        return None, "No file uploaded."
    content_type, content_string = contents.split(",")
    decoded = base64.b64decode(content_string)
    try:
        if filename.endswith(".csv"):
            df = pd.read_csv(io.StringIO(decoded.decode("utf-8")))
        elif filename.endswith((".xls", ".xlsx")):
            df = pd.read_excel(io.BytesIO(decoded))
        elif filename.endswith(".json"):
            df = pd.read_json(io.StringIO(decoded.decode("utf-8")))
        elif filename.endswith(".tsv"):
            df = pd.read_csv(io.StringIO(decoded.decode("utf-8")), sep="\t")
        else:
            return None, f"Unsupported file type: {filename}"
        return df, None
    except Exception as e:
        return None, f"Error reading file: {e}"

# ── 1. Parse upload → store ────────────────────────────────────────────────────

@callback(
    Output("store-df", "data"),
    Output("store-original-df", "data"),
    Output("upload-status", "children"),
    Input("upload-data", "contents"),
    State("upload-data", "filename"),
    prevent_initial_call=True,
)
def handle_upload(contents, filename):
    df, err = parse_upload(contents, filename)
    if err:
        msg = dmc.Alert(err, color="red", variant="filled",
                        icon=DashIconify(icon="tabler:alert-circle"))
        return no_update, no_update, msg

    json_data = df.to_json(date_format="iso", orient="split")
    ok = dmc.Alert(
        f"✓  Loaded {filename}  —  {len(df):,} rows × {len(df.columns)} columns",
        color="teal", variant="filled",
        icon=DashIconify(icon="tabler:circle-check"),
    )
    return json_data, json_data, ok


# ── 2. Render info bar + table ─────────────────────────────────────────────────

@callback(
    Output("data-info-bar", "children"),
    Output("table-container", "children"),
    Input("store-df", "data"),
)
def render_table(data):
    if not data:
        return no_update, no_update

    df = pd.read_json(io.StringIO(data), orient="split")
    n_rows, n_cols = df.shape
    n_null = int(df.isnull().sum().sum())
    numeric_cols = df.select_dtypes(include="number").shape[1]

    # Info badges
    info_bar = dmc.Group(
        gap="sm",
        children=[
            dmc.Badge(f"{n_rows:,} rows",    color="indigo", variant="light", size="lg"),
            dmc.Badge(f"{n_cols} columns",   color="cyan",   variant="light", size="lg"),
            dmc.Badge(f"{numeric_cols} numeric", color="teal", variant="light", size="lg"),
            dmc.Badge(
                f"{n_null} nulls",
                color="red" if n_null else "green",
                variant="light",
                size="lg",
            ),
        ],
    )

    # Table
    table = html.Div(
        style={
            "background": SURFACE,
            "borderRadius": "14px",
            "border": f"1px solid {BORDER}",
            "padding": "20px",
            "overflow": "hidden",
        },
        children=[
            dmc.Group(
                justify="space-between",
                mb="md",
                children=[
                    dmc.Text("Data Preview", fw=600, size="md", c=TEXT),
                    dmc.Text(
                        f"Showing up to 500 rows",
                        size="xs",
                        c=MUTED,
                    ),
                ],
            ),
            dash_table.DataTable(
                id="main-table",
                data=df.head(500).to_dict("records"),
                columns=[{"name": c, "id": c} for c in df.columns],
                page_size=20,
                page_action="native",
                sort_action="native",
                filter_action="native",
                style_table={"overflowX": "auto"},
                style_header={},
                style_cell={},
                style_data_conditional=[
                    {
                        "if": {"row_index": "odd"},
                        "background": CARD,
                    }
                ],
            ),
        ],
    )

    return info_bar, table

TOOL_IDS = [item["id"] for item in NAV_ITEMS]

    Output("tool-panel", "children"),
    Output("tool-panel", "style"),
    Output("store-active-tool", "data"),
    [Input(tid, "n_clicks") for tid in TOOL_IDS],
    State("store-df", "data"),
    State("store-active-tool", "data"),

    panel = html.Div(

            "padding": "24px",
            "width": "300px",
            "height": "fit-content",
            "position": "sticky",
            "top": "0",

                justify="space-between",

# ── 6. Filter ──────────────────────────────────────────────────────────────────

@callback(
    Output("store-df", "data", allow_duplicate=True),
    Output("upload-status", "children", allow_duplicate=True),
    Input("btn-filter", "n_clicks"),
    State("filter-col", "value"),
    State("filter-op", "value"),
    State("filter-val", "value"),
    State("store-df", "data"),
    prevent_initial_call=True,
)
def apply_filter(_, col, op, val, data):
    if not all([col, op, val, data]):
        raise PreventUpdate
    df = pd.read_json(io.StringIO(data), orient="split")
    try:
        if op == "eq":
            # try numeric first
            try:
                df = df[df[col] == float(val)]
            except (ValueError, TypeError):
                df = df[df[col].astype(str) == val]
        elif op == "neq":
            try:
                df = df[df[col] != float(val)]
            except (ValueError, TypeError):
                df = df[df[col].astype(str) != val]
        elif op == "gt":
            df = df[pd.to_numeric(df[col], errors="coerce") > float(val)]
        elif op == "lt":
            df = df[pd.to_numeric(df[col], errors="coerce") < float(val)]
        elif op == "contains":
            df = df[df[col].astype(str).str.contains(val, case=False, na=False)]
        msg = dmc.Alert(f"Filter applied — {len(df):,} rows remaining.",
                        color="teal", variant="filled",
                        icon=DashIconify(icon="tabler:filter"))
    except Exception as e:
        return no_update, dmc.Alert(str(e), color="red", variant="filled")

    return df.to_json(date_format="iso", orient="split"), msg


# ── 7. Sort ────────────────────────────────────────────────────────────────────

@callback(
    Output("store-df", "data", allow_duplicate=True),
    Output("upload-status", "children", allow_duplicate=True),
    Input("btn-sort", "n_clicks"),
    State("sort-col", "value"),
    State("sort-dir", "value"),
    State("store-df", "data"),
    prevent_initial_call=True,
)
def apply_sort(_, col, direction, data):
    if not all([col, direction, data]):
        raise PreventUpdate
    df = pd.read_json(io.StringIO(data), orient="split")
    df = df.sort_values(col, ascending=(direction == "asc"))
    msg = dmc.Alert(f"Sorted by '{col}' ({direction}).",
                    color="teal", variant="filled",
                    icon=DashIconify(icon="tabler:arrows-sort"))
    return df.to_json(date_format="iso", orient="split"), msg


# ── 8. Quick Stats ─────────────────────────────────────────────────────────────

@callback(
    Output("stats-output", "children"),
    Input("btn-stats", "n_clicks"),
    State("store-df", "data"),
    prevent_initial_call=True,
)
def show_stats(_, data):
    if not data:
        return dmc.Alert("No data loaded.", color="orange", variant="filled")
    df = pd.read_json(io.StringIO(data), orient="split")
    num = df.select_dtypes(include="number")
    if num.empty:
        return dmc.Alert("No numeric columns found.", color="orange", variant="filled")
    desc = num.describe().round(3)
    rows = []
    for stat in desc.index:

@callback(

    Output("upload-status", "children", allow_duplicate=True),
    Input("btn-fillna", "n_clicks"),
    State("fillna-col", "value"),
    State("fillna-val", "value"),
    State("store-df", "data"),
    prevent_initial_call=True,
)
def fill_nulls(_, col, val, data):
    if val is None or not data:
        raise PreventUpdate
    df = pd.read_json(io.StringIO(data), orient="split")
    # Try to coerce numeric fill value
    fill = val
    try:
        fill = float(val) if "." in val else int(val)
    except (ValueError, TypeError):
        pass

    if col == "__all__":
        before = int(df.isnull().sum().sum())
        df = df.fillna(fill)
        after = int(df.isnull().sum().sum())
    else:
        before = int(df[col].isnull().sum())
        df[col] = df[col].fillna(fill)
        after = int(df[col].isnull().sum())

    filled = before - after
    msg = dmc.Alert(f"Filled {filled} null(s) with '{fill}'.",
                    color="violet", variant="filled",
                    icon=DashIconify(icon="tabler:puzzle"))
    return df.to_json(date_format="iso", orient="split"), msg

def parse_upload(contents, filename):
    content_type, content_string = contents.split(",")
    decoded = base64.b64decode(content_string)
    try:
        if filename.endswith(".csv"):
            df = pd.read_csv(io.StringIO(decoded.decode("utf-8")))
        elif filename.endswith((".xlsx", ".xls")):
            df = pd.read_excel(io.BytesIO(decoded))
        else:
            return None
    except Exception:
        return None
    return df

def store_upload(contents, filename):
    if not contents:
        raise PreventUpdate
    df = parse_upload(contents, filename)
    if df is None:
        raise PreventUpdate

    records = df.to_json(orient="records", date_format="iso")
    cols = [{"label": c, "value": c} for c in df.columns]
    num_cols = [{"label": c, "value": c} for c in df.select_dtypes("number").columns]
    info = f"{filename}  ·  {len(df):,} rows  ·  {len(df.columns)} cols"
    return records, records, info, cols, cols, cols, num_cols

# ── Helpers ────────────────────────────────────────────────────────────────────

def parse_upload(contents, filename):
    content_type, content_string = contents.split(",")
    decoded = base64.b64decode(content_string)
    if filename.endswith(".csv"):
        return pd.read_csv(io.StringIO(decoded.decode("utf-8")))
    elif filename.endswith((".xls", ".xlsx")):
        return pd.read_excel(io.BytesIO(decoded))
    return None


def df_to_store(df):
    return df.to_json(date_format="iso", orient="split")


def store_to_df(data):
    return pd.read_json(io.StringIO(data), orient="split")


def build_table(df):
    return dash_table.DataTable(
        data=df.to_dict("records"),
        columns=[{"name": c, "id": c} for c in df.columns],
        page_size=15,
        style_table={"overflowX": "auto"},
        style_cell={"padding": "8px 12px", "fontFamily": "inherit", "fontSize": 13},
        style_header={
            "backgroundColor": "#f1f3f5",
            "fontWeight": "600",
            "borderBottom": "2px solid #dee2e6",
        },
        style_data_conditional=[
            {"if": {"row_index": "odd"}, "backgroundColor": "#f8f9fa"}
        ],
        sort_action="native",
        filter_action="native",
        export_format="csv",
    )

# 1. Upload → store
@callback(
    Output("raw-store", "data"),
    Output("working-store", "data"),
    Output("upload-status", "children"),
    Input("upload-data", "contents"),
    State("upload-data", "filename"),
    prevent_initial_call=True,
)
def handle_upload(contents, filename):
    df = parse_upload(contents, filename)
    if df is None:
        return None, None, "❌ Unsupported file type."
    stored = df_to_store(df)
    msg = f"✅ Loaded '{filename}' — {len(df):,} rows × {len(df.columns)} columns"
    return stored, stored, msg


# 2. Populate sidebar column selects whenever working store changes
@callback(
    Output("filter-column", "data"),
    Output("sort-column", "data"),
    Output("stats-column", "data"),
    Output("drop-columns", "data"),
    Input("working-store", "data"),
    prevent_initial_call=True,
)
def update_column_options(data):
    if not data:
        raise PreventUpdate
    df = store_to_df(data)
    opts = [{"value": c, "label": c} for c in df.columns]
    return opts, opts, opts, opts


# 3. Show summary cards
@callback(
    Output("summary-cards", "children"),
    Input("working-store", "data"),
    prevent_initial_call=True,
)
def show_summary(data):
    if not data:
        return []
    df = store_to_df(data)
    numeric_cols = df.select_dtypes("number").columns
    cards = [
        dmc.Card(
            withBorder=True, shadow="xs", radius="md", p="md",
            children=[
                dmc.Text(label, size="xs", c="dimmed", tt="uppercase", fw=600),
                dmc.Text(value, size="xl", fw=700),
            ],
        )
        for label, value in [
            ("Rows", f"{len(df):,}"),
            ("Columns", str(len(df.columns))),
            ("Numeric Cols", str(len(numeric_cols))),
            ("Missing Values", f"{df.isnull().sum().sum():,}"),
        ]
    ]
    return dmc.SimpleGrid(cols={"base": 2, "sm": 4}, children=cards)

def apply_filter(n, col, val, data):

# 8. Compute stats
@callback(
    Output("stats-output", "children"),
    Input("compute-stats-btn", "n_clicks"),
    State("stats-column", "value"),
    State("working-store", "data"),
    prevent_initial_call=True,
)
def compute_stats(n, col, data):
    if not n or not data or not col:
        raise PreventUpdate
    df = store_to_df(data)
    series = df[col]
    if pd.api.types.is_numeric_dtype(series):
        items = [
            ("Count",  f"{series.count():,}"),
            ("Mean",   f"{series.mean():.3f}"),
            ("Median", f"{series.median():.3f}"),
            ("Std",    f"{series.std():.3f}"),
            ("Min",    f"{series.min()}"),
            ("Max",    f"{series.max()}"),
        ]
    else:
        items = [
            ("Count",    f"{series.count():,}"),
            ("Unique",   f"{series.nunique():,}"),
            ("Top Value", str(series.mode().iloc[0]) if not series.mode().empty else "—"),
            ("Missing",  f"{series.isnull().sum():,}"),
        ]
    return dmc.Stack(
        gap=4,
        children=[
            dmc.Group(
                justify="space-between",
                children=[
                    dmc.Text(k, size="xs", c="dimmed"),
                    dmc.Text(v, size="xs", fw=600),
                ],
            )
            for k, v in items
        ],
    )
