def _make_csv(rows: int = 10) -> str:
    """Return path to a temporary CSV file."""
    df = pd.DataFrame({
        "name":   [f"Item {i}" for i in range(rows)],
        "value":  list(range(rows)),
        "score":  [round(i * 1.5, 2) for i in range(rows)],
        "flag":   [i % 2 == 0 for i in range(rows)],
        "nullable": [None if i % 3 == 0 else f"v{i}" for i in range(rows)],
    })
    tmp = tempfile.NamedTemporaryFile(suffix=".csv", delete=False, mode="w")
    df.to_csv(tmp.name, index=False)
    tmp.close()
    return tmp.name

class Page:
    """Thin helper that wraps WebDriver + WebDriverWait."""

    def __init__(self, driver: webdriver.Chrome, timeout: int = TIMEOUT):
        self.d = driver
        self.w = WebDriverWait(driver, timeout)

    # -- Navigation -----------------------------------------------------------

    def load(self, url: str = BASE_URL) -> None:
        self.d.get(url)
        self.wait_for_app()

    def wait_for_app(self) -> None:
        """Wait until the DataStudio app root is visible."""
        self.w.until(EC.presence_of_element_located((By.ID, "app-root")))
        # Also wait for React to finish hydrating (Dash quirk)
        time.sleep(0.5)

    # -- Elements -------------------------------------------------------------

    def find(self, by: By, value: str):
        return self.w.until(EC.presence_of_element_located((by, value)))

    def find_clickable(self, by: By, value: str):
        return self.w.until(EC.element_to_be_clickable((by, value)))

    def find_id(self, eid: str):
        return self.find(By.ID, eid)

    def click_id(self, eid: str) -> None:
        self.find_clickable(By.ID, eid).click()

    # -- Upload ---------------------------------------------------------------

    def upload_file(self, filepath: str) -> None:
        """
        Send the file path to the hidden <input type=file> inside dcc.Upload.
        Dash renders the real input inside the #upload-data container.
        """
        inp = self.w.until(
            EC.presence_of_element_located(
                (By.CSS_SELECTOR, "#upload-data input[type='file']")
            )
        )
        inp.send_keys(filepath)
        # Wait for the upload-status to update
        self.w.until(
            lambda d: d.find_element(By.ID, "upload-status").text.strip() != ""
            or d.find_element(By.ID, "grid-container").get_attribute("style") != "display: none;"
        )

    # -- Tools ----------------------------------------------------------------

    def open_tool(self, tool_id: str) -> None:
        """Click a sidebar nav tool button and wait for the panel to appear."""
        self.find_clickable(By.ID, tool_id).click()
        self.w.until(
            EC.presence_of_element_located((By.ID, "tool-panel-content"))
        )

    # -- Status / alerts ------------------------------------------------------

    def get_status_text(self) -> str:
        el = self.find_id("upload-status")
        return el.text.strip()

    def alert_visible(self, color_keyword: str = "") -> bool:
        """Check that a Mantine Alert is visible, optionally matching text."""
        try:
            alerts = self.d.find_elements(By.CSS_SELECTOR, "[data-mantine-alert]")
            if not alerts:
                # Fallback: look for any role=alert
                alerts = self.d.find_elements(By.CSS_SELECTOR, "[role='alert']")
            if color_keyword:
                return any(color_keyword.lower() in a.text.lower() for a in alerts)
            return len(alerts) > 0
        except Exception:
            return False

    def grid_visible(self) -> bool:
        try:
            gc = self.d.find_element(By.ID, "grid-container")
            style = gc.get_attribute("style") or ""
            return "none" not in style
        except Exception:
            return False

    def modal_open(self, modal_id: str) -> bool:
        try:
            modal = self.d.find_element(By.ID, modal_id)
            return modal.get_attribute("data-opened") == "true" or \
                   "mantine-Modal-content" in (modal.get_attribute("class") or "") or \
                   modal.is_displayed()
        except Exception:
            return False

        tmp = tempfile.NamedTemporaryFile(suffix=".txt", delete=False, mode="w")

                pass
            assert any(k in status + modal_text for k in ("unsupported", "error", "type", ".txt")), \
                "Expected an unsupported-file-type error"
        finally:
            os.unlink(tmp.name)

class TestMenuModals:
    """Hamburger menu → modals open correctly."""

    MODAL_MAP = {
        "menu-about":     "about-modal",
        "menu-shortcuts": "shortcuts-modal",
        "menu-contact":   "contact-modal",
        "menu-bug":       "bug-modal",
        "menu-changelog": "changelog-modal",
    }

    @pytest.fixture(autouse=True)
    def _setup(self, page):
        page.load()

    def _open_hamburger(self, page) -> None:
        """Open the hamburger / overflow menu button."""
        try:
            btn = page.find_clickable(By.ID, "hamburger-menu")
            btn.click()
        except Exception:
            # Fallback: look for a button with tabler:menu icon aria
            btn = page.find_clickable(
                By.CSS_SELECTOR, "[aria-label*='menu'], [id*='hamburger'], [id*='menu-btn']"
            )
            btn.click()
        time.sleep(0.4)

    @pytest.mark.parametrize("menu_id,modal_id", list(MODAL_MAP.items()))
    def test_menu_modal_opens(self, page, menu_id, modal_id):
        self._open_hamburger(page)
        try:
            page.find_clickable(By.ID, menu_id).click()
            time.sleep(0.6)
            # Check that *some* modal overlay is present in DOM
            overlays = page.d.find_elements(
                By.CSS_SELECTOR, ".mantine-Modal-overlay, [data-mantine-modal-root]"
            )
            modal_el = page.d.find_element(By.ID, modal_id)
            assert overlays or modal_el.is_displayed(), \
                f"Modal '{modal_id}' did not open after clicking '{menu_id}'"
        except Exception as exc:
            pytest.skip(f"Hamburger menu item '{menu_id}' not accessible: {exc}")


class TestResponsiveness:
    """Layout sanity at different viewport widths."""

    def _setup(self, page, sample_csv):
        page.load()
        page.upload_file(sample_csv)

        (1024, 768),   # laptop
        (768,  1024),  # tablet portrait
    ])
    def test_layout_at_viewport(self, driver, page, width, height):
        driver.set_window_size(width, height)
        time.sleep(0.5)

        driver.set_window_size(1400, 900)


class TestNoDataGuards:
