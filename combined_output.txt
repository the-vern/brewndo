"""
API Request Splitter
--------------------
Handles APIs with a 500,000 record limit by:
1. Checking the record count for a payload via the count endpoint
2. Splitting timestamps into smaller batches if the limit is exceeded
3. Recursively validating each batch before submitting
"""

import math
import logging
from typing import Any
import requests

logger = logging.getLogger(__name__)

MAX_RECORDS = 500_000


def check_record_count(count_url: str, payload: dict, session: requests.Session) -> int:
    """Call the count endpoint and return the number of records for this payload."""
    response = session.post(count_url, json=payload)
    response.raise_for_status()
    return response.json()["count"]


def submit_request(data_url: str, payload: dict, session: requests.Session) -> Any:
    """Submit a single API request and return the response data."""
    response = session.post(data_url, json=payload)
    response.raise_for_status()
    return response.json()


def split_timestamps(timestamps: list, num_chunks: int) -> list[list]:
    """Split a list of timestamps into `num_chunks` roughly equal parts."""
    chunk_size = math.ceil(len(timestamps) / num_chunks)
    return [timestamps[i : i + chunk_size] for i in range(0, len(timestamps), chunk_size)]


def fetch_all_records(
    count_url: str,
    data_url: str,
    payload: dict,
    timestamp_key: str = "timestamps",
    session: requests.Session = None,
) -> list[Any]:
    """
    Fetch all records for a payload, splitting by timestamps if the record
    count exceeds MAX_RECORDS (500,000).

    Args:
        count_url:      URL of the count endpoint  (e.g. "https://api.example.com/count")
        data_url:       URL of the data endpoint   (e.g. "https://api.example.com/data")
        payload:        Full request payload dict. Must contain a list of timestamps
                        under `timestamp_key`.
        timestamp_key:  The key in `payload` whose value is the list of timestamps.
                        Defaults to "timestamps".
        session:        Optional requests.Session (one is created if not provided).

    Returns:
        A flat list of all results across every batch request.
    """
    if session is None:
        session = requests.Session()

    timestamps = payload.get(timestamp_key)
    if not timestamps:
        raise ValueError(f"Payload must contain a non-empty list at key '{timestamp_key}'.")

    results = []
    # Use a stack to avoid deep recursion for very large timestamp lists.
    # Each item is a list of timestamps to process.
    pending_batches: list[list] = [timestamps]

    while pending_batches:
        current_timestamps = pending_batches.pop()
        batch_payload = {**payload, timestamp_key: current_timestamps}

        logger.info("Checking record count for batch of %d timestamp(s)…", len(current_timestamps))
        count = check_record_count(count_url, batch_payload, session)
        logger.info("  → %d records", count)

        if count <= MAX_RECORDS:
            # Safe to submit
            logger.info("  Submitting batch (%d timestamps, %d records).", len(current_timestamps), count)
            result = submit_request(data_url, batch_payload, session)
            results.append(result)
        else:
            # Need to split further; aim for two halves, but use math to be safe
            num_chunks = math.ceil(count / MAX_RECORDS)
            # We can't split further than one timestamp per chunk
            num_chunks = min(num_chunks, len(current_timestamps))

            if num_chunks <= 1:
                raise RuntimeError(
                    f"A single timestamp returns {count} records, which exceeds the "
                    f"{MAX_RECORDS:,} limit. Cannot split further."
                )

            logger.info(
                "  Count %d exceeds limit. Splitting %d timestamps into %d chunks.",
                count,
                len(current_timestamps),
                num_chunks,
            )
            chunks = split_timestamps(current_timestamps, num_chunks)
            # Push chunks onto the stack (reverse so the first chunk is processed first)
            pending_batches.extend(reversed(chunks))

    return results


# ---------------------------------------------------------------------------
# Example usage
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(levelname)s %(message)s")

    example_payload = {
        "timestamps": [
            "2024-01-01T00:00:00Z",
            "2024-01-02T00:00:00Z",
            "2024-01-03T00:00:00Z",
            "2024-01-04T00:00:00Z",
        ],
        "filters": {"status": "active"},
    }

    all_results = fetch_all_records(
        count_url="https://api.example.com/records/count",
        data_url="https://api.example.com/records",
        payload=example_payload,
        timestamp_key="timestamps",
    )

    print(f"Total batches returned: {len(all_results)}")
