# ── Helper: parse uploaded file ────────────────────────────────────────────────

def _human_size(n_bytes: int) -> str:
    for unit in ("B", "KB", "MB", "GB"):
        if n_bytes < 1024:
            return f"{n_bytes:.1f} {unit}"
        n_bytes /= 1024
    return f"{n_bytes:.1f} TB"


def parse_upload(contents, filename):
    """
    Decode base64 upload → (DataFrame | None, error_dict | None, warning_str | None).

    Returns a 3-tuple:
      df       – parsed DataFrame, or None on failure
      err      – dict with keys: title, detail, workarounds (list of str), or None
      warning  – human-readable warning string, or None
    """
    log.info("Upload received: filename=%s", filename)
    t0 = time.perf_counter()

    # ── Base64 decode ──────────────────────────────────────────────────────────
    try:
        _content_type, content_string = contents.split(",", 1)
        decoded = base64.b64decode(content_string)
    except Exception as exc:
        log.error("Base64 decode failed for %s: %s", filename, exc)
        return None, {
            "title": "Corrupt upload",
            "detail": f"The file data could not be decoded: {exc}",
            "workarounds": [
                "Ensure the file was not modified during transfer.",
                "Try uploading again; the browser may have sent a partial file.",
            ],
        }, None

    raw_bytes = len(decoded)
    log.info("Decoded %s → %s", filename, _human_size(raw_bytes))

    # ── Hard size limit ────────────────────────────────────────────────────────
    if raw_bytes > SIZE_LIMIT_MB * 1024 * 1024:
        msg = (f"{filename} is {_human_size(raw_bytes)}, which exceeds the "
               f"{SIZE_LIMIT_MB} MB limit.")
        log.warning("File too large, rejected: %s", msg)
        return None, {
            "title": "File too large",
            "detail": msg,
            "workarounds": [
                f"Split the file into chunks smaller than {SIZE_LIMIT_MB} MB.",
                "Filter or aggregate the data before exporting and uploading.",
                "For CSV files, upload every Nth row using: "
                "`df.iloc[::10].to_csv('sample.csv')` in Python.",
            ],
        }, None

    # ── Size warning (non-blocking) ────────────────────────────────────────────
    warning = None
    if raw_bytes > SIZE_WARN_MB * 1024 * 1024:
        warning = (f"{filename} is {_human_size(raw_bytes)} — processing may be slow. "
                   f"The preview is limited to {PREVIEW_ROWS:,} rows.")
        log.warning("Large file warning: %s", warning)

    # ── Extension check ────────────────────────────────────────────────────────
    ext = filename.rsplit(".", 1)[-1].lower() if "." in filename else ""
    if ext not in {"csv", "xls", "xlsx", "json", "tsv"}:
        log.error("Unsupported file extension: .%s (%s)", ext, filename)
        return None, {
            "title": f"Unsupported file type: .{ext}",
            "detail": f"DataStudio cannot read '.{ext}' files.",
            "workarounds": [
                "Accepted formats: CSV, XLSX, XLS, JSON, TSV.",
                "Convert the file in Excel (Save As) or with pandas: "
                "`df.to_csv('out.csv', index=False)`.",
            ],
        }, None

    # ── Parse ──────────────────────────────────────────────────────────────────
    try:
        if ext == "csv":
            # Chunked read for large files
            if raw_bytes > SIZE_WARN_MB * 1024 * 1024:
                log.info("Using chunked CSV reader for %s", filename)
                chunks = pd.read_csv(
                    io.StringIO(decoded.decode("utf-8", errors="replace")),
                    chunksize=CHUNK_ROWS,
                    low_memory=False,
                )
                df = pd.concat(chunks, ignore_index=True)
            else:
                df = pd.read_csv(io.StringIO(decoded.decode("utf-8", errors="replace")),
                                 low_memory=False)

        elif ext in {"xls", "xlsx"}:
            df = pd.read_excel(io.BytesIO(decoded))

        elif ext == "json":
            # Try records / split / default orientations
            raw_str = decoded.decode("utf-8", errors="replace")
            for orient in (None, "split", "records", "index"):
                try:
                    df = pd.read_json(io.StringIO(raw_str), orient=orient)
                    break
                except Exception:
                    continue
            else:
                raise ValueError("Could not parse JSON with any known orientation.")

        elif ext == "tsv":
            df = pd.read_csv(io.StringIO(decoded.decode("utf-8", errors="replace")),
                             sep="\t", low_memory=False)

        elapsed = time.perf_counter() - t0
        log.info(
            "Parsed %s in %.2fs → %d rows × %d cols",
            filename, elapsed, len(df), len(df.columns),
        )
        return df, None, warning

    except MemoryError:
        log.exception("MemoryError while parsing %s", filename)
        return None, {
            "title": "Out of memory",
            "detail": f"The server ran out of memory while reading {filename} "
                      f"({_human_size(raw_bytes)}).",
            "workarounds": [
                "Upload a smaller file or a random sample of rows.",
                "For CSVs: `pd.read_csv('file.csv').sample(100_000).to_csv('sample.csv')`",
                "Restart the app on a machine with more RAM.",
            ],
        }, None

    except UnicodeDecodeError as exc:
        log.error("Encoding error in %s: %s", filename, exc)
        return None, {
            "title": "Encoding error",
            "detail": f"Could not decode {filename} as UTF-8. ({exc})",
            "workarounds": [
                "Re-save the file with UTF-8 encoding (in Excel: Save As → CSV UTF-8).",
                "In Python: `df.to_csv('out.csv', encoding='utf-8', index=False)`",
            ],
        }, None

    except Exception as exc:
        log.error("Parse error for %s:\n%s", filename, tb)
        return None, {
            "title": "File parse error",
            "detail": str(exc),
            "workarounds": [
                "Ensure the file is not password-protected or corrupted.",
                "Try opening the file in Excel/a text editor to verify it looks correct.",
                "Check the server log (datastudio.log) for the full stack trace.",
            ],
        }, None


# ── Helper: build advanced error modal ────────────────────────────────────────

def make_error_modal_contents(title: str, detail: str, workarounds: list[str], is_blocked: bool = False):
    """
    Return (modal_title_div, modal_body_div, opened=True) for the persistent
    error modal.  Callers spread these across the three Output targets:
        error-modal-title, error-modal-body, error-modal.opened
    """
    icon_color = "#ff6b6b" if is_blocked else "#ffa94d"
    modal_title = dmc.Group(
        gap="xs",
        children=[
            DashIconify(icon="tabler:alert-triangle", width=22, color=icon_color),
            dmc.Text(title, fw=700, size="md", c=icon_color),
        ],
    )
    modal_body = dmc.Box([
        dmc.Text(detail, size="sm", c=TEXT, mb="md"),
        dmc.Divider(label="Possible workarounds", labelPosition="left", mb="sm"),
        dmc.Stack(
            gap="xs",
            children=[
                dmc.Group(
                    gap="xs",
                    align="flex-start",
                    children=[
                        DashIconify(icon="tabler:circle-arrow-right",
                                    width=16, color=ACCENT2),
                        dmc.Text(w, size="sm", c=MUTED, style={"flex": 1}),
                    ],
                )
                for w in workarounds
            ],
        ),
        dmc.Text(
            "Full details are written to datastudio.log.",
            size="xs", c=MUTED, mt="lg",
            style={"fontStyle": "italic"},
        ),
    ])
    return modal_title, modal_body, True

def _df_from_store(data: str) -> pd.DataFrame:
    """Deserialise a stored JSON string back to a DataFrame."""
    return pd.read_json(io.StringIO(data), orient="split")


def _df_to_store(df: pd.DataFrame) -> str:
    """Serialise a DataFrame to a JSON string for dcc.Store."""
    return df.to_json(date_format="iso", orient="split")

# ── 0. Client-side pre-flight size check ───────────────────────────────────────
# Triggered the instant the browser selects a file via dcc.Upload's file input.
# We use a MutationObserver injected on page load to attach a capturing "change"
# listener to the hidden <input type="file"> BEFORE Dash's own listener fires.
# If the file is too large we clear input.value so Dash never reads the contents,
# then write the error payload into store-size-blocked / store-size-warning via
# a second clientside_callback that polls window.__dsSize every 300 ms.

# Step A — inject the file-input interceptor once on page load
app.clientside_callback(
    """
    function(_) {
        var LIMIT_MB = """ + str(SIZE_LIMIT_MB) + """;
        var WARN_MB  = """ + str(SIZE_WARN_MB)  + """;
        var PREVIEW  = """ + str(PREVIEW_ROWS)  + """;

        function fmt(mb) { return mb.toFixed(1) + ' MB'; }

        function handleFileChange(e) {
            var input = e.target;
            var file  = input.files && input.files[0];
            if (!file) return;

            var sizeMB = file.size / (1024 * 1024);

            if (sizeMB > LIMIT_MB) {
                // Wipe the selection — Dash will never see contents
                input.value = '';
                e.stopImmediatePropagation();
                window.__dsSize = {
                    blocked: true,
                    title:   'File too large (' + fmt(sizeMB) + ')',
                    detail:  '"' + file.name + '" is ' + fmt(sizeMB) +
                             ', which exceeds the ' + LIMIT_MB +
                             ' MB limit. The file was NOT uploaded.',
                    workarounds: [
                        'Split the file into parts smaller than ' + LIMIT_MB + ' MB.',
                        'Sample rows first: df.sample(50000).to_csv("sample.csv")',
                        'Use a database or streaming pipeline for very large data.'
                    ],
                    tick: (window.__dsSize ? (window.__dsSize.tick || 0) : 0) + 1
                };
            } else if (sizeMB > WARN_MB) {
                window.__dsSize = {
                    blocked: false,
                    title:   'Large file (' + fmt(sizeMB) + ')',
                    detail:  '"' + file.name + '" is ' + fmt(sizeMB) +
                             '. Processing may be slow; preview limited to ' +
                             PREVIEW.toLocaleString() + ' rows.',
                    workarounds: [
                        'Sample the data first for faster exploration.',
                        'Use the Export tool to save a filtered/smaller file.'
                    ],
                    tick: (window.__dsSize ? (window.__dsSize.tick || 0) : 0) + 1
                };
            } else {
                window.__dsSize = { blocked: false, clear: true, tick: Date.now() };
            }
        }

        function attachTo(input) {
            if (input.__dsAttached) return;
            input.__dsAttached = true;
            // capture=true → our handler runs before Dash's handler
            input.addEventListener('change', handleFileChange, true);
        }

        // Attach to any already-rendered inputs and watch for new ones
        document.querySelectorAll('#upload-data input[type="file"]')
                .forEach(attachTo);
        var obs = new MutationObserver(function() {
            document.querySelectorAll('#upload-data input[type="file"]')
                    .forEach(attachTo);
        });
        obs.observe(document.body, { childList: true, subtree: true });

        return window.dash_clientside.no_update;
    }
    """,
    Output("store-size-blocked", "data"),          # dummy output — never actually written here
    Input("_page-load-interval", "n_intervals"),
    prevent_initial_call=False,
)

# Step B — poll window.__dsSize every 300 ms and push results into the two stores
app.clientside_callback(
    """
    function(_) {
        var d = window.__dsSize;
        if (!d || d.tick === window.__dsSizeSeen) {
            return [window.dash_clientside.no_update, window.dash_clientside.no_update];
        }
        window.__dsSizeSeen = d.tick;
        if (d.clear) {
            return [false, window.dash_clientside.no_update];
        }
        return [
            d.blocked,
            { title: d.title, detail: d.detail, workarounds: d.workarounds }
        ];
    }
    """,
    Output("store-size-blocked", "data", allow_duplicate=True),
    Output("store-size-warning", "data"),
    Input("_poll-interval", "n_intervals"),
    prevent_initial_call=True,
)


# ── 0b. Render size warning / block modal from store ──────────────────────────

@callback(
    Output("error-modal-title", "children"),
    Output("error-modal-body",  "children"),
    Output("error-modal",       "opened", allow_duplicate=True),
    Output("upload-status",     "children", allow_duplicate=True),
    Output("store-modal-was-size-block", "data"),
    Input("store-size-warning", "data"),
    State("store-size-blocked", "data"),
    prevent_initial_call=True,
)
def show_size_modal(warning_data, blocked):
    if not warning_data:
        raise PreventUpdate

    title      = warning_data.get("title", "File size issue")
    detail     = warning_data.get("detail", "")
    workarounds = warning_data.get("workarounds", [])

    log.warning("Size pre-check: blocked=%s title=%s", blocked, title)

    modal_title, modal_body, opened = make_error_modal_contents(
        title, detail, workarounds, is_blocked=blocked
    )

    alert_color = "red" if blocked else "yellow"
    alert_icon  = "tabler:alert-circle" if blocked else "tabler:alert-triangle"
    alert = dmc.Alert(
        detail, title=title, color=alert_color, variant="filled",
        icon=DashIconify(icon=alert_icon),
    )

    return modal_title, modal_body, opened, alert, blocked

# ── 1. Parse upload → store ────────────────────────────────────────────────────

@callback(
    Output("store-df",           "data"),
    Output("store-original-df",  "data"),
    Output("store-history",      "data",              allow_duplicate=True),
    Output("upload-status",      "children"),
    Output("error-modal-title",  "children",          allow_duplicate=True),
    Output("error-modal-body",   "children",          allow_duplicate=True),
    Output("error-modal",        "opened",            allow_duplicate=True),
    Output("upload-card",        "style"),
    Output("reset-btn-wrapper",  "style"),
    Output("tool-panel",         "opened",            allow_duplicate=True),
    Output("store-active-tool",  "data",              allow_duplicate=True),
    Input("upload-data",         "contents"),
    State("upload-data",         "filename"),
    State("store-size-blocked",  "data"),
    prevent_initial_call=True,
)
def handle_upload(contents, filename, size_blocked):
    # contents is None when reset clears the upload component — ignore it
    if contents is None:
        raise PreventUpdate

    log.info("handle_upload triggered for: %s (size_blocked=%s)", filename, size_blocked)

    # Client-side already showed the modal — skip processing entirely
    if size_blocked:
        log.warning("handle_upload blocked by size pre-check for: %s", filename)
        raise PreventUpdate

    df, err, warning = parse_upload(contents, filename)

    if err:
        log.warning("Upload rejected — %s: %s", err["title"], err["detail"])
        alert = dmc.Alert(
            err["detail"], title=err["title"], color="red", variant="filled",
            icon=DashIconify(icon="tabler:alert-circle"),
        )
        mt, mb, opened = make_error_modal_contents(
            err["title"], err["detail"], err["workarounds"], is_blocked=True
        )
        return no_update, no_update, no_update, alert, mt, mb, opened, UPLOAD_CARD_VISIBLE, no_update, False, None

    json_data = _df_to_store(df)

    status_children = []
    if warning:
        status_children.append(
            dmc.Alert(
                warning, title="Large file", color="yellow", variant="light",
                icon=DashIconify(icon="tabler:alert-triangle"), mb="xs",
            )
        )
    status_children.append(
        dmc.Alert(
            f"✓  Loaded {filename}  —  {len(df):,} rows × {len(df.columns)} columns",
            color="teal", variant="filled",
            icon=DashIconify(icon="tabler:circle-check"),
        )
    )

    return json_data, json_data, [], dmc.Box(status_children), no_update, no_update, False, {"display": "none"}, {}, False, None


# ── 2. Render info bar + table ─────────────────────────────────────────────────

@callback(
    Output("data-info-bar", "children"),
    Output("table-container", "children"),
    Input("store-df", "data"),

def render_table(data):
    if not data:
        return no_update, no_update

    t0 = time.perf_counter()
    df = _df_from_store(data)
    n_rows, n_cols = df.shape
    n_null = int(df.isnull().sum().sum())
    numeric_cols = df.select_dtypes(include="number").shape[1]
    log.info("render_table: %d rows × %d cols, %d nulls", n_rows, n_cols, n_null)

    # Info badges
    info_bar = dmc.Group(
        gap="sm",
        children=[
            dmc.Badge(f"{n_rows:,} rows",       color="indigo", variant="light", size="lg"),
            dmc.Badge(f"{n_cols} columns",      color="cyan",   variant="light", size="lg"),
            dmc.Badge(f"{numeric_cols} numeric", color="teal",  variant="light", size="lg"),
            dmc.Badge(
                f"{n_null} nulls",
                color="red" if n_null else "green",
                variant="light",
                size="lg",
            ),
        ],
    )

    preview_df = df.head(PREVIEW_ROWS)

# ── 6. Filter ──────────────────────────────────────────────────────────────────

@callback(
    Output("store-df",          "data",     allow_duplicate=True),
    Output("store-history",     "data",     allow_duplicate=True),
    Output("upload-status",     "children", allow_duplicate=True),
    Output("error-modal-title", "children", allow_duplicate=True),
    Output("error-modal-body",  "children", allow_duplicate=True),
    Output("error-modal",       "opened",   allow_duplicate=True),
    Input("btn-filter", "n_clicks"),
    State("filter-col", "value"),
    State("filter-op",  "value"),
    State("filter-val", "value"),
    State("store-df",   "data"),
    State("store-history", "data"),
    prevent_initial_call=True,
)
def apply_filter(_, col, op, val, data, history):
    if not all([col, op, val, data]):
        raise PreventUpdate
    log.info("apply_filter: col=%s op=%s val=%s", col, op, val)
    df = _df_from_store(data)
    try:
        if op == "eq":
            try:
                df = df[df[col] == float(val)]
            except (ValueError, TypeError):
                df = df[df[col].astype(str) == val]
        elif op == "neq":
            try:
                df = df[df[col] != float(val)]
            except (ValueError, TypeError):
                df = df[df[col].astype(str) != val]
        elif op == "gt":
            df = df[pd.to_numeric(df[col], errors="coerce") > float(val)]
        elif op == "lt":
            df = df[pd.to_numeric(df[col], errors="coerce") < float(val)]
        elif op == "contains":
            df = df[df[col].astype(str).str.contains(val, case=False, na=False)]
        log.info("Filter applied — %d rows remaining", len(df))
        msg = dmc.Alert(f"Filter applied — {len(df):,} rows remaining.",
                        color="teal", variant="filled",
                        icon=DashIconify(icon="tabler:filter"))
        new_history = (history or []) + [data]
        new_history = new_history[-2:]
        return _df_to_store(df), new_history, msg, no_update, no_update, no_update
    except Exception as e:
        log.exception("Filter error: %s", e)
        mt, mb, opened = make_error_modal_contents(
            "Filter error", str(e),
            ["Verify the column type matches the operator (e.g. don't use > on text).",
             "Check that the value field is not empty."],
        )
        return no_update, no_update, dmc.Alert(str(e), color="red", variant="filled"), mt, mb, opened

@callback(
    Output("stats-output", "children"),
    Input("btn-stats", "n_clicks"),
    State("store-df", "data"),
    prevent_initial_call=True,
)
def show_stats(_, data):
    if not data:
        return dmc.Alert("No data loaded.", color="orange", variant="filled")
    log.info("show_stats triggered")
    df = _df_from_store(data)
    num = df.select_dtypes(include="number")
    if num.empty:
        return dmc.Alert("No numeric columns found.", color="orange", variant="filled")
    desc = num.describe().round(3)
    rows = []
    for stat in desc.index:
        row = [dmc.TableTd(stat, fw=600, c=MUTED)]
        for col in desc.columns:
            row.append(dmc.TableTd(str(desc.loc[stat, col])))
        rows.append(dmc.TableTr(row))

    header_cells = [dmc.TableTh("")] + [dmc.TableTh(c) for c in desc.columns]

    return dmc.Table(
        style={"fontSize": "11px"},
        children=[
            dmc.TableThead(dmc.TableTr(header_cells)),
            dmc.TableTbody(rows),
        ],
        striped=True,
        highlightOnHover=True,
        withTableBorder=True,
        withColumnBorders=True,
    )

# ── 11. Fill nulls ─────────────────────────────────────────────────────────────

@callback(
    Output("store-df", "data", allow_duplicate=True),
    Output("store-history", "data", allow_duplicate=True),
    Output("upload-status", "children", allow_duplicate=True),
    Input("btn-fillna", "n_clicks"),
    State("fillna-col", "value"),
    State("fillna-val", "value"),
    State("store-df", "data"),
    State("store-history", "data"),
    prevent_initial_call=True,
)
def fill_nulls(_, col, val, data, history):
    if val is None or not data:
        raise PreventUpdate
    log.info("fill_nulls: col=%s val=%r", col, val)
    df = _df_from_store(data)
    # Try to coerce numeric fill value
    fill = val
    try:
        fill = float(val) if "." in val else int(val)
    except (ValueError, TypeError):
        pass

    if col == "__all__":
        before = int(df.isnull().sum().sum())
        df = df.fillna(fill)
        after = int(df.isnull().sum().sum())
    else:
        before = int(df[col].isnull().sum())
        df[col] = df[col].fillna(fill)
        after = int(df[col].isnull().sum())

    filled = before - after
    msg = dmc.Alert(f"Filled {filled} null(s) with '{fill}'.",
                    color="violet", variant="filled",
                    icon=DashIconify(icon="tabler:puzzle"))
    new_history = ((history or []) + [data])[-2:]
    return _df_to_store(df), new_history, msg
